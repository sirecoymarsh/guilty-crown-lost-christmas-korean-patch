//■ 읽음: 장면을 끝까지 본 기록(원작 이야기 루프의 $AllRead 와 같은 Conquest) 또는 검수용 모두 열기.
//   $KS_SEEN = 실제로 읽음(결말 제목은 이것일 때만), $KS_OPEN = 열 수 있음
// Conquest 는 원작처럼 변수에 받는다. if(Conquest(...)) 로 바로 쓰면 그 뒤 함수 호출에서 넘긴 변수가 깨진다(11-02 숨긴 시험).
function KsIsOpen($ko_n)
{
	$KS_OPEN=false;
	$KS_SEEN=false;
	$ko_s=$KS_FILE[$ko_n];
	$ko_f="nss/"+$ko_s;
	$ko_r=Conquest($ko_f,$ko_s,null);
	if($ko_r){$KS_SEEN=true;$KS_OPEN=true;}
	if(#KS_ALLOPEN){$KS_OPEN=true;}
}

//■ 넘겨 볼 수 있는 마지막 단계: 열린 장면이 있는 가장 뒤 단계. 하나도 없으면 1.
function KsMaxPage()
{
	$KS_MAXP=1;
	$kmp_p=1;
	while($kmp_p<=$KS_NPAGE){
		$kmp_j=0;
		$kmp_c=$KS_PC[$kmp_p];
		while($kmp_j<$kmp_c){
			$kmp_n=$KS_PS[$kmp_p]+$kmp_j;
			KsIsOpen($kmp_n);
			if($KS_OPEN){$KS_MAXP=$kmp_p;}
			$kmp_j++;
		}
		$kmp_p++;
	}
}

function KsAllToggle()

function KsPageStep($kps_d)
{
	$KS_PAGE=$KS_PAGE+$kps_d;
	if($KS_PAGE>$KS_MAXP){$KS_PAGE=1;}
	if($KS_PAGE<1){$KS_PAGE=$KS_MAXP;}
	$KS_LOOP=false;
}

function KsPick($kpk_i)
{
	Request("@ks_se_ok",Play);
	$KS_SEL=$KS_PS[$KS_PAGE]+$kpk_i;
	$KS_LOOP=false;
}

//■ 열린 앞·뒤 장면(단계를 넘어서도)
function KsStep($kst_d)
{
	$kst_m=$KS_SEL+$kst_d;
	$kst_go=true;
	while($kst_go){
		if($kst_m<0){$kst_go=false;}
		else if($kst_m>=$KS_N){$kst_go=false;}
		else{
			KsIsOpen($kst_m);
			if($KS_OPEN){$KS_SEL=$kst_m;$KS_PAGE=$KS_PG[$kst_m];$kst_go=false;}
			else{$kst_m=$kst_m+$kst_d;}
		}
	}
}
