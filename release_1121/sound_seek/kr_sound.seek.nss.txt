chapter main
{
	$SYSTEM_menu_lock = true;
	$SYSTEM_backlog_lock = true;

function KuCover()
{
	Delete("ku_cover");
	CreateTexture("ku_cover",30000,0,0,SCREEN);
}

function KuSounds()
{
	CreateSound("ku_se_ok",SE,"sound/se/SE_긘긚긡��_긄긌긚긣깋_뙂믦.ogg");
	CreateSound("ku_se_end",SE,"sound/se/SE_긘긚긡��_긄긌긚긣깋_뢎뿹.ogg");
	SetAlias("ku_se_ok","ku_se_ok");
	SetAlias("ku_se_end","ku_se_end");
}

function KuText($ut_nm,$ut_p,$ut_x,$ut_y,$ut_w,$ut_h,$ut_t)
{
	CreateText($ut_nm,$ut_p,$ut_x,$ut_y,$ut_w,$ut_h,$ut_t);
	Request($ut_nm,NoLog);
	Request($ut_nm,PushText);
}

function KuClear()
{
	Delete("ku_l*");
	Delete("ku_r*");
	Delete("KU_*");
}

// 다시 그리는 그림은 이름 둘을 번갈아 쓴다(같은 이름을 한 프레임에 지우고 만들면 새것이 사라질 때가 있었다).

{

// 11-24 사용자 「사운드 모드 반응이 굉장히 느린데?」 - 커서를 옮길 때마다 글이나 그림을 새로 만들면 곡이 나오는 동안 1~2초씩 멈췄다
// (프로세스 둘이 같이 만들 때 가장 심함). 그래서 막대 갱신도 이 프로세스가 같이 돌리고, 풀이 그림(kr_hint_*.png)은 목록을 그릴 때
// 이 프로세스가 한 번 다 만들어 숨겨 두었다가 커서를 옮기면 보이기·숨기기만 한다(본 대본이 만든 그림은 프로세스에서 못 건드려
// 「사운드」 그림도 여기서 만든다).
function KuHintRow()
{
	$KUH_R=-1;$KUH_SH=0;$KUH_ON=true;
	$KUH_G5=false;if(#KS_ALLOPEN){$KUH_G5=true;}
	if(!$KUH_G5){$kc_r=Conquest("nss/lx0206.nss","lx0206.nss",null);if($kc_r){$KUH_G5=true;}}
	$KUH_G14=false;if(#KS_ALLOPEN){$KUH_G14=true;}
	if(!$KUH_G14){$kc_r=Conquest("nss/lx0603.nss","lx0603.nss",null);if($kc_r){$KUH_G14=true;}}
	KuHintMake();
	if($KUH_PA==1){$KUH_PA=0;$khPN="ku_lhprocB";}else{$KUH_PA=1;$khPN="ku_lhprocA";}
	CreateProcess($khPN,150,0,0,"KuHintLoop");
	Request($khPN,Start);
}

// $KUH_RUN: 고리가 도는 동안 true. 다시 그리기(KuList)는 $KUH_ON 을 끄고 이것이 꺼질 때까지 기다린 뒤 지운다
// - 돌고 있는 프로세스를 Delete 로 지우면 엔진이 그 스레드를 100+1500+1500ms 기다리다 끊어 3.1초 멈췄다(다른 세션 「사운드 모드 성능」 11-24).
function KuHintLoop()
{
	$KUH_RUN=true;
	while($KUH_ON){
		KuHintTick();
		if($KUS_ON){KuSeekTick();}
		Wait(16);
	}
	$KUH_RUN=false;
}

function KuHintTick()
{
	$khX=-1;$khY=-1;
	CursorPosition($khX,$khY);
	$khR=0;
	if($khY>=58&&$khY<498){
		$khD=$khY-58;$khQ=$khD/44;$khK=Integer($khQ);$khM=$khK*44;$khO=$khD-$khM;
		if($khO<26){
			if($khX>=26&&$khX<502){$khR=$khK+1;}
			if($khX>=524&&$khX<1000){$khR=$khK+11;}
		}
	}
	if($khR>19){$khR=0;}
	if($khR!=$KUH_R){
		$KUH_R=$khR;
		$khN=0;
		if($khR==1){if($KU_OK1){$khN=1;}}
		if($khR==2){if($KU_OK2){$khN=2;}}
		if($khR==3){if($KU_OK3){$khN=3;}}
		if($khR==4){if($KU_OK4){$khN=4;}}
		if($khR==5){if($KU_OK5){$khN=5;}}
		if($khR==6){if($KU_OK6){$khN=6;}}
		if($khR==7){if($KU_OK7){$khN=7;}}
		if($khR==8){if($KU_OK8){$khN=8;}}
		if($khR==9){if($KU_OK9){$khN=9;}}
		if($khR==10){if($KU_OK10){$khN=10;}}
		if($khR==11){if($KU_OK11){$khN=11;}}
		if($khR==12){if($KU_OK12){$khN=12;}}
		if($khR==13){if($KU_OK13){$khN=13;}}
		if($khR==14){if($KU_OK14){$khN=14;}}
		if($khR==15){if($KU_OK15){$khN=15;}}
		if($khR==16){if($KU_OK16){$khN=16;}}
		if($khR==17){if($KU_OK17){$khN=17;}}
		if($khR==18){if($KU_OK18){$khN=18;}}
		if($khR==19){if($KU_OK19){$khN=19;}}
		if($khN!=$KUH_SH){
			KuHintShow();
			KuHintHide();
			$KUH_SH=$khN;
		}
	}
}
