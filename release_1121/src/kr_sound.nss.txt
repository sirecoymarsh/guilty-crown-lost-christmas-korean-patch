#include "nss/function.nss"
#include "nss/function_move.nss"
#include "nss/function_select.nss"

chapter main
{
	$SYSTEM_menu_lock = true;
	$SYSTEM_backlog_lock = true;
	$KU_NOW=-1;
	SetVolume("@*", 500, 0, NULL);
	Wait(500);
	KuCalc();
	$KU_RUN=true;
	while($KU_RUN){
		KuList();
	}
	CreateColor("kz_black",31000,0,0,1024,576,BLACK);
	Fade("kz_black",0,0,null,true);
	Fade("kz_black",300,1000,null,true);
	SetVolume("@kubgm", 800, 0, NULL);
	Wait(800);
	Delete("@kubgm");
	Delete("*");
}

function KuCover()
{
	Delete("ku_cover");
	CreateTexture("ku_cover",30000,0,0,SCREEN);
}

function KuSounds()
{
	CreateSound("ku_se_ok",SE,"sound/se/SE_긘긚긡��_긄긌긚긣깋_뙂믦.ogg");
	CreateSound("ku_se_end",SE,"sound/se/SE_긘긚긡��_긄긌긚긣깋_뢎뿹.ogg");
	SetAlias("ku_se_ok","ku_se_ok");
	SetAlias("ku_se_end","ku_se_end");
}

function KuText($ut_nm,$ut_p,$ut_x,$ut_y,$ut_w,$ut_h,$ut_t)
{
	CreateText($ut_nm,$ut_p,$ut_x,$ut_y,$ut_w,$ut_h,$ut_t);
	Request($ut_nm,NoLog);
	Request($ut_nm,PushText);
}

function KuClear()
{
	Delete("ku_l*");
	Delete("ku_r*");
	Delete("KU_*");
}

//■ 곡 바꾸기: 앞 곡을 줄여 지우고 새로 튼다(같은 이름 하나만 쓴다)
function KuFade()
{
	SetVolume("@kubgm", 300, 0, NULL);
	Wait(300);
	Delete("@kubgm");
}
