//■ダイアログツイッター
function Dialogtwitter()
{
	if(!$SYSTEM_twitter_lock){
		call_chapter nss/sys_twitter.nss;
	}
}

//■ダイアログフルスクリーン
function DialogFull()
{
	if(!#SYSTEM_window_full_lock){
		#SYSTEM_window_full=!#SYSTEM_window_full;
		#SYSTEM_window_full_lock=false;
		Wait(300);
		$SYSTEM_keydown_f=false;
	}
}

//■ダイアログクローズ
function DialogClose()
{
	DialogGo();

	$InDialog=true;
	if($Title_Config||$Title_Load){call_chapter nss/sys_close.nss->main2;}
	else{call_chapter nss/sys_close.nss;}
	$InDialog=false;

	DialogReturn();
}

//■ダイアログタイトル
function DialogTitle()
{
	DialogGo();

	$InDialog=true;
	call_chapter nss/sys_reset.nss;
	$InDialog=false;

	DialogReturn();
}

//■ダイアログバックセレクト
function DialogBack()
{
	DialogGo();

	$InDialog=true;
	call_chapter nss/sys_backselect.nss;
	$InDialog=false;

	DialogReturn();
}

	$SYSTEM_buttondown_close=false;
}
