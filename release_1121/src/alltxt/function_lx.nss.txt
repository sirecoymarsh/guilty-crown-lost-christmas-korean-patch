function CreateOratorioTxtSet($OratorioTxtNut,Pri,$CreateOratorio_X,$CreateOratorio_Y,$OTextTexture)
{
	SetFont("Malgun Gothic", 30, FFFFFF, 000000, MEDIUM, RIGHTDOWN);
	CreateText($OratorioTxtNut, Pri, $CreateOratorio_X, $CreateOratorio_Y, Auto, Auto, $OTextTexture);
	SetAlias($OratorioTxtNut,$OratorioTxtNut);
	Fade($OratorioTxtNut, 0, 0, null, true);
	Request($OratorioTxtNut, PushText);
	Request($OratorioTxtNut, NoLog);
	Request($OratorioTxtNut, NoIcon);

	$TATE=576;

//ｘ取得
	$CTO_EXX = ImageHorizon($OratorioTxtNut);
//ｙ取得
	$CTO_EXY = ImageVertical($OratorioTxtNut);
//ｘ取得÷２
	$CTO_EXX2 = $CTO_EXX-1024;
//ｙ取得÷２
	$CTO_EXY2 = $CTO_EXY-$TATE;//768;
//ｘ取得÷４
	$CTO_EXX2 = $CTO_EXX2/2;
//ｙ取得÷４
	$CTO_EXY2 = $CTO_EXY2/2;
//Center取得
	$CTO_EX_Center = -($CTO_EXX2);
//Middle取得
	$CTO_EX_Middle = -($CTO_EXY2);
//Center取得
	$CTO_EX_Center2 = 1024-$CTO_EXX;
//Right取得
	$CTO_EX_Center3 = $CTO_EX_Center2;
	$CTO_EX_Center4 = -($CTO_EX_Center3);
	$CTO_EX_Center2 = $CTO_EX_Center2/2;
//Middle取得
	$CTO_EX_Middle2 = $TATE-$CTO_EXY;//768;
//Right取得
	$CTO_EX_Middle3 = $CTO_EX_Middle2;
	$CTO_EX_Middle4 = -($CTO_EX_Middle3);
	$CTO_EX_Middle2 = $CTO_EX_Middle2/2;

//InBottom取得
	$CTO_EX_InBottom0 = $CTO_EXY-$TATE;//768;
	$CTO_EX_InBottom = -($CTO_EX_InBottom0);

	SetAlias($OratorioTxtNut, $OratorioTxtNut);

	if($CreateOratorio_X=="Center"){
//横幅＝1024
		if($CTO_EXX == 1024){
			Move($OratorioTxtNut, 0, 0, @0, null, true);
//横幅＜1024
		}else if($CTO_EXX < 1024){
			Move($OratorioTxtNut, 0, $CTO_EX_Center2, @0, null, true);
//横幅＞1024
		}else{
			Move($OratorioTxtNut, 0, $CTO_EX_Center, @0, null, true);
		}
	}else if($CreateOratorio_X=="center"){
		if($CTO_EXX == 1024){
			Move($OratorioTxtNut, 0, 0, @0, null, true);
		}else if($CTO_EXX < 1024){
			Move($OratorioTxtNut, 0, $CTO_EX_Center2, @0, null, true);
		}else{
			Move($OratorioTxtNut, 0, $CTO_EX_Center, @0, null, true);
		}
	}else if($CreateOratorio_X=="InLeft"){
			Move($OratorioTxtNut, 0, 0, @0, null, true);
	}else if($CreateOratorio_X=="InRight"){
//横幅＝1024
		if($CTO_EXX == 1024){
			Move($OratorioTxtNut, 0, 0, @0, null, true);
//横幅＜1024
		}else if($CTO_EXX < 1024){
			Move($OratorioTxtNut, 0, $CTO_EX_Center3, @0, null, true);
//横幅＞1024
		}else{
			Move($OratorioTxtNut, 0, $CTO_EX_Center3, @0, null, true);
		}
	}else{
//		$CreateOratorio_X = $CreateOratorio_X-1030;
		Move($OratorioTxtNut, 0, 0, @0, null, true);
		Move($OratorioTxtNut, 0, $CreateOratorio_X, @0, null, true);
	}


//縦幅関連
	if($CreateOratorio_Y=="Middle"){
		if($CTO_EXY == $TATE){//768
			Move($OratorioTxtNut, 0, @0, 0, null, true);
		}else if($CTO_EXX <= $TATE){//768
			Move($OratorioTxtNut, 0, @0, $CTO_EX_Middle2, null, true);
		}else{
			Move($OratorioTxtNut, 0, @0, $CTO_EX_Middle, null, true);
		}
	}else if($CreateOratorio_Y=="middle"){
		if($CTO_EXY == $TATE){//768
			Move($OratorioTxtNut, 0, @0, 0, null, true);
		}else if($CTO_EXX <= $TATE){//768
			Move($OratorioTxtNut, 0, @0, $CTO_EX_Middle2, null, true);
		}else{
			Move($OratorioTxtNut, 0, @0, $CTO_EX_Middle, null, true);
		}
	}else if($CreateOratorio_Y=="InBottom"){//768
		if($CTO_EXY != $TATE){Move($OratorioTxtNut, 0, @0, $CTO_EX_InBottom, null, true);}else{Move($OratorioTxtNut, 0, @0, 0, null, true);}
	}else if($CreateOratorio_Y=="inBottom"){//768
		if($CTO_EXY != $TATE){Move($OratorioTxtNut, 0, @0, $CTO_EX_InBottom, null, true);}else{Move($OratorioTxtNut, 0, @0, 0, null, true);}
	}else if($CreateOratorio_Y=="inbottom"){//768
		if($CTO_EXY != $TATE){Move($OratorioTxtNut, 0, @0, $CTO_EX_InBottom, null, true);}else{Move($OratorioTxtNut, 0, @0, 0, null, true);}
	}else{
		Move($OratorioTxtNut, 0, @0, 0, null, true);
		Move($OratorioTxtNut, 0, @0, $CreateOratorio_Y, null, true);
	}
}
}
