function HAFade(優先度, $レベル数, _待ち)
{
	Delete("頭痛prc");

	if($レベル数==0){$HACurrent=$HALevel;}
	else{$HACurrent=$レベル数;}
	$HAON=2;
	
	if($HACurrent==5){$img="ef蒼葉黒強頭痛d.jpg";}
	else if($HACurrent==4){$img="ef蒼葉黒強頭痛c.jpg";}
	else if($HACurrent==3){$img="ef蒼葉黒強頭痛b.jpg";}
	else if($HACurrent==2){$img="ef蒼葉黒強頭痛a.jpg";}
	else {$img="ef蒼葉黒強頭痛.jpg";$HACurrent=1;}
	$img="cg/ef/"+$img;
	CreateTextureEXmul("頭痛", 優先度, Center, Middle, $img);
	Zoom("@頭痛",0,1100,1100,null,false);Request("@頭痛",Smoothing);

//	CreateProcess("頭痛prc",優先度,0,0,"HAProcess");
//	Request("頭痛prc",Start);

	$vol=$HACurrent*100+500;
	CreateSE("SE01頭痛","se人体_頭痛_痛む01");
	MusicStart("SE01頭痛",0,$vol,0,1000,null,false);

		Fade("@頭痛", 0, 1000, null, false);//WaitKey();
		Shake("@頭痛", 200, 0,0, 3,5, 500,Dxl1,false);
		DrawTransition("@頭痛", 200, 0, 1000, 500, Dxl1, "cg/data/circle_01_00_1.png", 400);
		//DrawTransition("@頭痛", 1000, 1000, 0, 500, Axl1, "cg/data/circle_01_00_1.png", false);
			$HATIME=$HACurrent*100+500;
			Fade("@頭痛", $HATIME, 0, null, _待ち);
			$HAON=0;
			Request("@頭痛",Disused);

}
//頭痛エフェクト開始プロセス（ループあり）
//―――――――――――――――
//HAStart($レベル)
//―――――――――――――――
//１～５で手動設定
//０で自動化
function HAStart(優先度, $レベル数)
{
	Delete("頭痛prc");

	if($レベル数==0){$HACurrent=$HALevel;}
	else{$HACurrent=$レベル数;}
	$HAON=1;
	
	if($HACurrent==5){$img="ef蒼葉黒強頭痛d.jpg";}
	else if($HACurrent==4){$img="ef蒼葉黒強頭痛c.jpg";}
	else if($HACurrent==3){$img="ef蒼葉黒強頭痛b.jpg";}
	else if($HACurrent==2){$img="ef蒼葉黒強頭痛a.jpg";}
	else {$img="ef蒼葉黒強頭痛.jpg";$HACurrent=1;}
	$img="cg/ef/"+$img;
	CreateTextureEXmul("頭痛", 優先度, Center, Middle, $img);

	CreateProcess("頭痛prc",優先度,0,0,"HAProcess");
	Request("頭痛prc",Start);
}
//―――――――――――――――
//HAAuoStart()
//―――――――――――――――
function HAAuoStart(優先度)
{
	HAStart(優先度, $HALevel);
}
function HAProcess()
{
	$HATIME=(6-$HACurrent)*150+250;
	$HAFADE=$HACurrent*100;
	begin:
	
	while($HAON>1)
	{
		Wait(200);
		Fade("@頭痛", 0, 1000, null, false);
		DrawTransition("@頭痛", 200, 0, 1000, 500, Dxl1, "cg/data/circle_01_00_1.png", 400);
		//DrawTransition("@頭痛", 1000, 1000, 0, 500, Axl1, "cg/data/circle_01_00_1.png", false);
		if($HAON==2){
			$HATIME=$HACurrent*100+500;
			Fade("@頭痛", $HATIME, 0, null, true);
			$HAON=0;
		}else{
			Fade("@頭痛", $HATIME, $HAFADE, null, true);
		
		}
	}
}

{
	$HAON=0;

	FadeDelete("@頭痛",1000,null,false);//Request("@頭痛",Disused);//Delete("@頭痛");
	Delete("頭痛prc");
}

function CreateMaskSetEX($MaskNut, $優先度, $Xpos, $Ypos, $MaskDataName, $子継承, $MaskOverData)
{
	CreateMaskSet();
	if($MaskOverData!=""){CreateTextureEX($OverNutName, $優先度, $Xpos, $Ypos, $OverData);}

}

function CreateMaskSet()
{
	$MaskData = "cg/mask/" + $MaskDataName + ".png";
	CreateMask($MaskNut, 0, $Xpos, $Ypos, $MaskData, $子継承);
	SetAlias($MaskNut,$MaskNut);

	$OverNutName = $MaskNut + "枠";//Line
	if($MaskOverData!=""){$OverData = "cg/mask/" + $MaskOverData + ".png";}
}

//---------------------------------------------------------------//
..OpenTateMask
//---------------------------------------------------------------//
function OpenTateMask($MaskNut, $所要時間, $水平倍率, $テンポ, $待ち)
{
	$wc=$水平倍率/1000*362;
	$w1=(1024-$wc)/2;
	$xL=$w1-7;
	$xR=$w1+$wc-7;
	
	$nutL=$MaskNut+"枠左";
	$nutR=$MaskNut+"枠右";

	Move($nutL, $所要時間, $xL, @0,$テンポ,false);
	Move($nutR, $所要時間, $xR, @0,$テンポ,false);
	Zoom($MaskNut,$所要時間,$水平倍率,1000,$テンポ,$待ち);
}
//---------------------------------------------------------------//
..CloseTateMask
//---------------------------------------------------------------//
function CloseTateMask($MaskNut, $所要時間, $待ち)
{
	$nut=$MaskNut+"*";
	FadeDelete($nut, $所要時間, null, false);
	$nut=$MaskNut+"枠左";
	Move($nut, $所要時間, 505, @0,$テンポ,false);
	$nut=$MaskNut+"枠右";
	Move($nut, $所要時間, 505, @0,$テンポ,false);
	Zoom($MaskNut,$所要時間,0,1000,$テンポ,$待ち);
}
