	#BEGIN_TIME=Time();

	}

function TitleSelect()
{
	CreateColor("tbk",1100,center,middle,1024,576,BLACK);
	Fade("tbk",0,0,null,false);
	while(!$TitleSelect)
	{
		select
		{
			if($TitleMovieCount){break;}

			case Title_Liner{
				CreateSE("決定サウンド",$TitleClickSE);
				SoundPlay("決定サウンド",0,1000,false);

				Wait(100);

				#SYSTEM_window_full = false;
				System("OPEN:http://www.nitroplus.co.jp/secret/gclostxmas_staffcomments2012/");
			}

			case Title_Start{
				TitleMovieIn();
				$TitleSelect = 1;
			}

			case Title_Scenario{
				CreateSE("決定サウンド",$TitleClickSE);
				SoundPlay("決定サウンド",0,650,false);
				$TitleSelect=3;
			}

			case Title_Sound{
				CreateSE("決定サウンド",$TitleClickSE);
				SoundPlay("決定サウンド",0,650,false);
				$TitleSelect=4;
			}

			case Title_LOAD{
//				if(#SaveFirst){
					TitleMovieIn();
					CreateSE("決定サウンド",$TitleClickSE);
					SoundPlay("決定サウンド",0,650,false);
					$Title_Load=true;
//				}
			}

			case タイトル特別{
				TitleMovieIn();
				CreateSE("決定サウンド",$TitleClickSE3);
				SoundPlay("決定サウンド",0,650,false);
				$TitleSelect=2;
			}

			case Title_CG{
				TitleMovieIn();
				CreateSE("決定サウンド",$TitleClickSE);
				SoundPlay("決定サウンド",0,650,false);
				$Title_Extra01=true;
			}

			case Title_Config{
				TitleMovieIn();
				CreateSE("決定サウンド",$TitleClickSE);
				SoundPlay("決定サウンド",0,850,false);
				$Title_Config=true;
			}

			case Title_Exit
			{
				CreateSE("決定サウンド",$TitleClickSE);
				SoundPlay("決定サウンド",0,1000,false);

				Wait(100);

				TitleMovieIn();

				$Title_Exit=true;
				call_chapter nss/sys_close.nss;
				$Title_Exit=false;

				TitleMovieOut();
			}

			//★キーダウン系
			if($SYSTEM_keydown_f){
				if(!#SYSTEM_window_full_lock){
					#SYSTEM_window_full=!#SYSTEM_window_full;
					#SYSTEM_window_full_lock=false;
					Wait(300);
					$SYSTEM_keydown_f=false;
				}
			}
		}

		if($TitleSelect== 1){
			CreateSE("決定サウンド",$TitleClickSE2);
			Request("決定サウンド", Lock);
			SoundPlay("決定サウンド",0,850,false);

			Request("*/MouseUsual/hit",UnLock);
			Delete("*/MouseUsual/hit");
			SetVolume("@*", 5000, 0, NULL);
			Fade("タイトルカラー", 5000, 1000, null, true);

			$GameName="lx0010.nss";
		}else if($Title_Load){
			Fade("tbk",150,1000,null,true);
			call_chapter nss/sys_load.nss;
			CreateMovieEXadd("TITLE_絵Snow", 999, Center, Middle, true, true, "dx/mvsnow01.ngs");
			Request("TITLE_絵Snow", Smoothing);
			Zoom("TITLE_絵Snow", 0, 2000, 2000, null, false);
			Fade("TITLE_絵Snow", 1000, 1000, null, false);
			Fade("tbk",150,0,null,true);
			$Title_Load=false;
			TitleMovieOut();
		}else if($Title_Extra01){
			Fade("tbk",150,1000,null,true);
			call_chapter nss/extra_cglib.nss;
			CreateMovieEXadd("TITLE_絵Snow", 999, Center, Middle, true, true, "dx/mvsnow01.ngs");
			Request("TITLE_絵Snow", Smoothing);
			Zoom("TITLE_絵Snow", 0, 2000, 2000, null, false);
			Fade("TITLE_絵Snow", 1000, 1000, null, false);
			Fade("tbk",150,0,null,true);
			$Title_Extra01=false;
			TitleMovieOut();
		}else if($Title_Extra02){
			call_chapter nss/extra_soundlib.nss;
			$Title_Extra02=false;
			TitleMovieOut();
		}else if($Title_Config){
			Fade("tbk",150,1000,null,true);
			call_chapter nss/sys_config.nss;
			CreateMovieEXadd("TITLE_絵Snow", 999, Center, Middle, true, true, "dx/mvsnow01.ngs");
			Request("TITLE_絵Snow", Smoothing);

			Fade("TITLE_絵Snow", 1000, 1000, null, false);
			Fade("tbk",150,0,null,true);
			$Title_Config=false;
			TitleMovieOut();
		}
	}
