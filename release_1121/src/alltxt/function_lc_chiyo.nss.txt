function tCreateChoices($nut_hd,$x,$y,$txt)
{
	CreateChoice($nut_hd);
	$nut_name=$nut_hd+"/MouseUsual/img";
//	$w=300;$h=30;$xplus=10;$yplus=-8;
	CreateColor($nut_name,150, $x,$y, $w,$h, #999999);
	$nut_name=$nut_hd+"/MouseOver/img";
	CreateColor($nut_name,150, $x,$y, $w,$h, FFFFFF);
	Fade($nut_name,0,0,null,false);
	$nut_name=$nut_hd+"/MouseClick/img";
	CreateColor($nut_name,150, $x,$y, $w,$h, FFFFFF);
	Fade($nut_name,0,0,null,false);
	$nut_name=$nut_hd+"/txt";
	$tx=$x+$xplus;
	$ty=$y+$yplus;
	CreateText($nut_name,150, $tx,$ty, auto,auto, $txt);
	Request($nut_name,PushText);
	Request($nut_name,NoLog);
}
