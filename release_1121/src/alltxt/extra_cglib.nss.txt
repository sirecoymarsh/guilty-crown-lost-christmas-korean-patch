function SetXtraThumbnailXY($num)
{
	$nx=$num%5;
	$my_x=($nx * 200) + 16;

	$ny=Integer($num/5);
	$my_y=($ny * 124) + 36;
}
//ＣＧサムネイルのファイル名取得
function SetCGThumFile($num,$dir,$sub)
{
	$anth=$num;
	$filename=$CG_ARY[$anth][$sub];
//CreateText("ナット名", 120,0,0, auto,auto,"$anth=<VALUE name=$anth>:$ThumFile=<VALUE name=$ThumFile>");
	//ChkThumFileName($num);
	$ext=".jpg";

	if(Substr($filename,0,2)=="ev"){$fldname="ev/";}
	else if(Substr($filename,0,2)=="bg"){$fldname="bg/";}
	else {$fldname="";}

	$ThumFile="cg"+$dir+$fldname+$filename+$ext;

}

//ＣＧサムネイル表示
function show_cg($cgno)
{
	$CG_NO=$cgno;
	$EVPG_NUM=0;
	
	$EVPG_MAX_NUM=0;
	SetMaxEvNum($CG_NO);
	//CreateText("ナット名1", 2000,0,400, auto,auto,"$CG_NO=<VALUE name=$CG_NO>");
	//CreateText("ナット名2", 2000,0,420, auto,auto,"$EVPG_MAX_NUM=<VALUE name=$EVPG_MAX_NUM>");
	$SYSTEM_r_button_down=false;
	$CG_LIB=true;
	while($CG_LIB)
	{
		//メニューサムネイル上下左右矢印キー封じ用カラselect
		CreateWindow("cgwin",2050, 0,0, 1024,768,true);
		SetAlias("cgwin","cgwin");
		select
		{
			$EVPG_NUM++;
			if($EVPG_NUM>$EVPG_MAX_NUM){
				//Request("chk_prs",Stop);
				Fade($nut,300,0,null,true);
				Delete($nut);
				$CG_LIB=false;
				break;
			}else{
				if($EVPG_NUM>1){$prevnut=$nut;}
				$sub_num=$EVPG_NUM-1;
				//evファイル名取得
				$IMG_SCRL=0;
				if(VariableValue(#,$CG_ARY[$CG_NO][$sub_num])){
					SetCGThumFile($CG_NO,"/",$sub_num);
					ChkScrl($CG_NO);
				}else{
					$ThumFile="cg/sys/extra/cglib/noimage.png";
				}
				$nut="@cgwin/"+"evnum_"+$EVPG_NUM;
				
				CreateTexture($nut,2050,center,0,$ThumFile);
				Fade($nut,0,0,null,true);
				
				if($CG_NO==42){
					Rotate($nut,0,0,180,0,null,false);
					Zoom($nut,0,1050,1050,null,false);
				}
				if($IMG_SCRL>0){
					$imgWidth =ImageHorizon($nut);
					$imgHeight=ImageVertical($nut);
					
					if($IMG_SCRL==1){//上から
						$sub=576-$imgHeight;
					}else if($IMG_SCRL==2){//下から
						$sub=576-$imgHeight;
						Move($nut,0,@0,$sub,null,false);
					}else if($IMG_SCRL==3){//左から
						$sub=1024-$imgWidth;
						Move($nut,0,0,@0,null,false);
					}else if($IMG_SCRL==4){//右から
						$sub=1024-$imgWidth;
						Move($nut,0,$sub,@0,null,false);
					}
				}
				
				FadeWaitKey($nut,300);//表示中キー待ち
				if($EVPG_NUM>1){Delete($prevnut);}
				
				//スクロールmove
				if($IMG_SCRL>0){
					if($sub<0){$scrltime=$sub*-10;}
					else{$scrltime=$sub*10;}
					if($IMG_SCRL==1){		MoveWaitKey($nut,$scrltime,@0,$sub);}
					else if($IMG_SCRL==2){	MoveWaitKey($nut,$scrltime,@0,0);}
					else if($IMG_SCRL==3){	MoveWaitKey($nut,$scrltime,$sub,0);}
					else if($IMG_SCRL==4){	MoveWaitKey($nut,$scrltime,0,0);}
				}
				
				WaitKey();
			}
		}
		$SYSTEM_r_button_down=false;

		//Delete("chk_prs");
		Delete("cgwin");
	}
}
function FadeWaitKey($_nut,$_tm)
{
	Fade($_nut,$_tm,1000,null,false);
	WaitKey($_tm);
	Fade($_nut,0,1000,null,true);
}
function MoveWaitKey($_nut,$_tm,$_x,$_y)
{
	WaitKey();
	Move($_nut,$_tm, $_x,$_y, null,false);
	WaitKey($_tm);
	Move($_nut,0, $_x,$_y, null,true);
}
