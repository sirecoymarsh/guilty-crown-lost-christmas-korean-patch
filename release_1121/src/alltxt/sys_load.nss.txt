chapter main
{
	//★ロックの場合
	if($SYSTEM_load_lock){
		return;
	}

	//★前提変数
	if($Title_Load){
		//★：ゲームスピード
		#play_speed_plus2 = #SYSTEM_play_speed;
		#SYSTEM_play_speed=3;
	}

	Wait(16);
	
	//★超速からの変遷用
	$SYSTEM_keydown_esc=false;
	$SYSTEM_buttondown_close=false;

	$SYSTEM_keydown_l = false;

	//■準備「スクリーン」//※右クリックから来ていなければ
	if(!$SYSTEM_menu_enable&&!$PLACE_title){
		CreateTexture("video",10000,center,middle,VIDEO);
		CreateTexture("SaveScreen", 200000, 0, 0, VIDEO);
	}else{
		CreateTexture("SaveScreen", 200000, 0, 0, SCREEN);
	}
	Delete("@TITLE_絵Snow");
	Wait(1);

	//行間調整
	$SYSTEM_text_margin_column=0;
	$SYSTEM_text_margin_row=0;

	//デリート初期化
	$SaveDelete=false;

	//初回無し
	$SaveFiest=false;

	//■定義「共通素材」
	SaveReady();

	//■アニメーション定義
	//CreateLoadAnime();

	//■開始「ロード画面」
	$SYSTEM_menu_load_enable=true;
	while($SYSTEM_menu_load_enable){

		//▼定義
		SaveSet();

		//▼描画
		SaveFade();

		//▼選択肢
		SaveSelect();
	}

	Request("セーブ_終了",Play);

	//★終了動作「描画」
	CreateTexture("システム背景SL", 20100, 0, 0, SCREEN);

	Delete("WND");
	Delete("Save*");
	Delete("@Save*");
	Request("@TITLE_絵Snow",Resume);
	Fade("システム背景SL",300,0,null,true);

	if($Title_Load){
		#SYSTEM_play_speed = #play_speed_plus2;
	}

	$SYSTEM_mousewheel_up=false;
	$SYSTEM_mousewheel_down=false;

	$SYSTEM_menu_load_enable=false;
}

function do_save($SaveCount)
{
	$SaveGroval=$SaveCount+((#SavePage-1)*$SaveMaxNum);
	$save=true;

..	//▼確認「新規か上書きか？」
	if(!ExistSave($SaveGroval)){
		return;
	}else{
		if(!#no_ask){
			SaveDialog();
//			SaveDialogTrial();
		}else{
			Request("セーブ_記録",Play);
		}
	}


..	//▼読込「ロード開始」
	if($save){
		if(!$SaveDelete){
			CreateColor("BLACK",99999999,center,middle,1024,768,BLACK);
			Fade("BLACK", 0, 0, null, true);
//			CreateTexture("MSG_load",99999999,362,334,"cg/sys/dialog/読み込んでいます.png");
			Fade("BLACK", 300, 1000, null, true);
	
			$SYSTEM_menu_enable=false;
			$SYSTEM_menu_load_enable=false;
	
			//★：ゲームスピード
			if($Title_Load){
				#SYSTEM_play_speed = #play_speed_plus2;
			}
	
			DeleteSaveFile(9999);
	
			#InitCV=false;
			#PreSaveFlag=false;
			#START_TIME=Time();
			Save(0);
	
			Load($SaveGroval);
		}else{
			DeleteSaveFile($SaveGroval);

			SaveThumbnail();
		}
	}

	//■終了共通演出
	DialogEnd();

	$OnDialog=false;
}
