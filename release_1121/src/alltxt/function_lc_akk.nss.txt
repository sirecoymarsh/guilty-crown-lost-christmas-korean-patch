//画面を移動しながら巻き上がる煙：定義

//カメラナット,煙の優先度,Ｘ座標＠指定,Ｙ座標＠指定,スケール,ファイル
function CreatePassSmoke($CPSName,$CPSPri,$CPSX,$CPSY,$CPSZ,$CPSFil)
{
	$CPSCamNut = "$CPSName";
	CreateCameraEX($CPSCamNut, 0, 0, 0);

	$CPSTexNut = "@" + "$CPSName" + "/" + "絵演粉上煙";
	CreateTextureEX($CPSTexNut, $CPSPri, 1024, 0, $CPSFil);
	SetVertex($CPSTexNut, center, 480);

	$CPSXSHalfcale = 0 - (ImageHorizon($CPSTexNut)/2);
	Move($CPSTexNut, 0, $CPSXSHalfcale, -409, null, true);

//	Position($CPSTexNut, $CPSXPOS, $CPSXPOS);
	Move($CPSTexNut, 0, $CPSX, $CPSY, null, true);//指定した位置に移動

	$CPSZSX = $CPSZ*0.6;
	$CPSZSY = $CPSZ*0.02;
	Zoom($CPSTexNut, 0, $CPSZS, $CPSZSY, null, true);//0.6
	Fade($CPSTexNut, 0, 0, null, true);

	$CPSCamPri = 1000 - $CPSPri;
	MoveCamera($CPSCamNut, 0, @0, @0, $CPSCamPri, null, true);

}

//画面を移動しながら巻き上がる煙：再生
//カメラナット,画面を横切る速度,移動先Ｘ座標＠指定,移動先Ｙ座標＠指定,スケール,テンポ
function PassSmoke($PSNut,$PSTime,$PSXEd,$PSYEd,$PSZEd,$PSTen)
{

	$PSTexNut = "$PSNut" + "/" + "絵演粉上煙";
	Fade($PSTexNut, 0, 1000, null, true);

	$PSZSX = $PSZEd*0.4;
	$PSZSY = $PSZEd*1.1;
	Zoom($PSTexNut, 150, $PSZSX, $PSZSY, Dxl3, false);

	Wait(30);
	Shake($PSTexNut, 150, 8, 0, 0, 0, 1000, null, false);
	MoveCamera($PSNut, $PSTime, $PSXEd, $PSYEd, @0, $PSTen, false);
	Wait(100);

	Zoom($PSTexNut, 1000, $PSZSX, $PSZSY, null, false);

	$PSYDM = $PSZEd*0.03;
	$PSYDMS = "@" + "$PSYDM";
	Move($PSTexNut, 1800, @0, $PSYDMS, null, false);

	Fade($PSTexNut, 300, 0, null, false);

}

//画面を移動しながら巻き上がる煙：再設置

{
	$RPSTexNut = "$RPSNut" + "/" + "絵演粉上煙";
	$RPSXSHalfcale = 0 - (ImageHorizon($RPSTexNut)/2);

	$RPSZSY = $RPSZ*0.02;
	Zoom($RPSTexNut, 0, 600, 50, null, true);

	MoveCamera($RPSNut, 0, 0, 0, @0, null, true);
}

function VacuumBullet($VCNut)
{
	Request($VCNut, Stop);
	Move($VCNut, 0, -300, -300, null, true);
	$RAKR = (Random(12)*30);
	Rotate($VCNut, 0, @0, @0, $RAKR, null,true);

	$RAKXSet = 100+(Random(80)*10);
	$RAKYSet = 50+(Random(40)*10);
	$RAKX = "@" + "$RAKXSet";
	$RAKY = "@" + "$RAKYSet";
	Move($VCNut, 0, $RAKX, $RAKY, null, true);

	Shake($VCNut, 200, 2, 4, 0, 0, 1000, null, false);

	$RAKZ = 1000 + (Random(5)*100);
	Zoom($VCNut, 200, $RAKZ, $RAKZ, Dxl2, false);
	Fade($VCNut, 0, 1000, null, true);
	Wait(50);
	Move($VCNut, 200, 212, 12, Axl1, false);
	Wait(50);
	Fade($VCNut, 100, 0, null, false);
}

//ミサイルを生成
function CreateMissile($MMNut,$MMPri,$CMXS,$CMYS,Zoom)
{
	//CreateText($MMNut, 0, 1024, 0, 4000, 4000, null);
	CreateColor($MMNut, 0, 1024, 0, 60, 60, "000000");
	Request($MMNut, AddRender);
	SetAlias($MMNut,$MMNut);
	SetVertex($MMNut, 0, middle);
	Fade($MMNut, 0, 0, null, true);

	$MMSmokeNut="@"+$MMNut+"/"+"_Smoke";
	CreateTextureEXadd($MMSmokeNut, $MMPri, 1024, 0, "cg/ef/ef_Missile01Smoke.png");
	SetBlur($MMSmokeNut, true, 3, 200, 200, false);
	SetVertex($MMSmokeNut, 0, middle);
	$YScale=0-(ImageVertical($MMSmokeNut)/2)+2000;
	Move($MMSmokeNut, 0, 0, $YScale, null, true);

	$MMBaseNut="@"+$MMNut+"/"+"_Base";
	$MMPri=$MMPri+10;
	CreateTextureEX($MMBaseNut, $MMPri, 1024, 0, "cg/ef/ef_Missile01.png");
	SetVertex($MMBaseNut, 0, middle);
	$YScale=0-(ImageVertical($MMBaseNut)/2)+2000;
	Move($MMBaseNut, 0, 0, $YScale, null, true);

	Zoom($MMNut, 0, Zoom, Zoom, null, true);
	Move($MMNut, 0, @-1024, @-2000, null, true);//基本位置に設定

	Position($MMNut, $CMX, $CMY);
	$CMXPos2=$CMXS-$CMX;
	$CMXPos=$CMXPos2;
	$CMYPos2=$CMYS-$CMY-2000;
	$CMYPos=$CMYPos2;
	Move($MMNut, 0, @$CMXPos, @$CMYPos, null, true);//設定位置に移動

	Fade($MMBaseNut, 0, 1000, null, true);
	Fade($MMBaseNut, 0, 1000, null, true);

}
