function CreateLoopFade($LoopTextureName,優先度,$LT_DrawOn,$LT_Render,画像パス１,画像パス２,画像パス３){

	$LoopTextureName1 = $LoopTextureName + "_1";
	$LoopTextureName2 = $LoopTextureName + "_2";
	$LoopTextureName3 = $LoopTextureName + "_3";

	CreateTextureEX($LoopTextureName1, 優先度, Center, Middle, 画像パス１);
	CreateTextureEX($LoopTextureName2, 優先度, Center, Middle, 画像パス２);
	CreateTextureEX($LoopTextureName3, 優先度, Center, Middle, 画像パス３);

	Request($LoopTextureName1, Smoothing);
	Request($LoopTextureName2, Smoothing);
	Request($LoopTextureName3, Smoothing);

	Request($LoopTextureName1, $LT_Render);
	Request($LoopTextureName2, $LT_Render);
	Request($LoopTextureName3, $LT_Render);

	Fade($LoopTextureName1, 0, 1000, null, true);
	Fade($LoopTextureName2, 0, 1000, null, true);
	Fade($LoopTextureName3, 0, 1000, null, true);

	if($LT_DrawOn){
		DrawTransition($LoopTextureName1, 0, 300, 300, 800, null, "cg/data/slide_06_00_1.png", true);
		DrawTransition($LoopTextureName2, 0, 300, 300, 800, null, "cg/data/slide_01_00_0.png", true);
		DrawTransition($LoopTextureName3, 0, 300, 300, 800, null, "cg/data/slide_01_00_1.png", true);
	}

	CreateProcess($LoopTextureName, 150, 0, 0, "CreateLoopFadePro");
	SetAlias($LoopTextureName,$LoopTextureName);

}

//食べ終わった茶碗をセット
function Yuru_Eat_BowlSet()
{

//フウリの後ろに重なっていくお茶碗の生成用変数を設定
	$EatBowlWhile=0;
	$EatBowlX_Def = 360;
	$EatBowlY_Def = 352;
	$EatBowlLevel_Def = 0;

	$EatBowlX = $EatBowlX_Def;
	$EatBowlY = $EatBowlY_Def;
	$EatBowlLevel = $EatBowlLevel_Def;

	//まずはフウリの後ろに出るお茶碗　４２個分を作成
	while(42>$EatBowlWhile){


		$EatBowlLevel = $EatBowlLevel+1;
		$EatBowlNum1=Integer($EatBowlWhile/10);
		$EatBowlNum2=Integer($EatBowlWhile%10);

		$EatBowlNut="ゆる窓/EatBowl"+$EatBowlNum1+"_"+$EatBowlNum2;

		CreateTexture($EatBowlNut, 99, $EatBowlX, $EatBowlY, "cg/yur/yurフウリ大食い_茶碗.png");
		SetAlias($EatBowlNut, $EatBowlNut);
		Fade($EatBowlNut, 0, 0, null, true);



		$EatBowlX = $EatBowlX + 80;

		if($EatBowlLevel>=6){

			$EatBowlX = $EatBowlX_Def - 30;
			$EatBowlX_Def = $EatBowlX;
			$EatBowlY = $EatBowlY - 35;

			$EatBowlLevel = $EatBowlLevel_Def;

		}

	$EatBowlWhile = $EatBowlWhile + 1;

	}

//フウリの前に重なっていくお茶碗の生成用変数を設定
//	$EatBowlWhile=0;
	$EatBowlX_Def = 300;
	$EatBowlY_Def = 400;
	$EatBowlLevel_Def = 0;

	$EatBowlX = $EatBowlX_Def;
	$EatBowlY = $EatBowlY_Def;
	$EatBowlLevel = $EatBowlLevel_Def;

	//フウリの前に重なっていくお茶碗　１００個分を作成
	while(100>$EatBowlWhile){


		$EatBowlLevel = $EatBowlLevel+1;
		$EatBowlNum1=Integer($EatBowlWhile/10);
		$EatBowlNum2=Integer($EatBowlWhile%10);

		$EatBowlNut="ゆる窓/EatBowl"+$EatBowlNum1+"_"+$EatBowlNum2;

		CreateTexture($EatBowlNut, 105, $EatBowlX, $EatBowlY, "cg/yur/yurフウリ大食い_茶碗.png");
//		Request($EatBowlNut, Smoothing);
		SetAlias($EatBowlNut, $EatBowlNut);
		Fade($EatBowlNut, 0, 0, null, true);
		Zoom($EatBowlNut, 0, 1200, 1200, null, true);



		$EatBowlX = $EatBowlX + 80;

		if($EatBowlLevel>=6){

			$EatBowlX = $EatBowlX_Def;
			$EatBowlX_Def = $EatBowlX;
			$EatBowlY = $EatBowlY - 38;

			$EatBowlLevel = $EatBowlLevel_Def;

		}

	$EatBowlWhile = $EatBowlWhile + 1;

	}

}

function manasong_start($manasong_vol)
{
	CreateSound("manasong", SE, "sound/se/se特殊_雰囲気_マナ歌声01.ogg");
	SetAlias("manasong","manasong");
	SetVolume("manasong", 0, $manasong_vol, null);

	CreateProcessEX("manasong_pronut", "manasong_pro");
	Request("manasong_pronut", Start);

}

function manasong_volchange($manasong_vol)
{
}

function manasong_pro()
{

	while(1){

		SetVolume("@manasong", 0, $manasong_vol, null);
		Request("@manasong", Play);
		WaitPlay("@manasong");
		Wait(1000);
	}

}

function manasong_delete($manasong_fade){

	Request("@manasong_pronut", Stop);
	SetVolume("@manasong", $manasong_fade, 0, null);
	Request("@manasong", Disused);
	Request("@manasong_pronut", Disused);

}
