function ArScroogeStop_Create($araStop_Nut,$araStop_pri,$backYposi,$ScroX,$Sczoom,$araScrooge_eye,$araSmoke_on,$araScrooge_back){


//変数指定ナット名
	$araDash_Nut = $araStop_Nut + "_dash";
	$araBack_Nut = $araStop_Nut + "_back";
	$araBackback_Nut = $araStop_Nut + "_backback";
	$araSmoke_Nut = $araStop_Nut + "_smoke";
	$araMovie_Nut = $araStop_Nut + "_movie";
	$ara_Nut_All = "@" + $araStop_Nut + "*";
	$araBack_All = "@" + $araBack_Nut + "*";

//変数指定優先度
	$araStop_pri2 = $araStop_pri + 1;
	$araStop_pri_back = $araStop_pri - 5;//背景
	$araStop_pri_backback = $araStop_pri_back + 1;//背景裏
	$araStop_pri_smoke = $araStop_pri + 6;//煙
	$araStop_pri_move = $araStop_pri + 5;//ムービー

//変数指定ファイル名
	$araScrooge_back_file =  "cg/bg/l/"+ $araScrooge_back + "_l.jpg";

	if($araScrooge_eye == true){
		CreateTextureEX($araDash_Nut, $araStop_pri2, Center, Middle, "cg/ev/x/ev9002スクルージダッシュ汎用_a_x01.png");
	}else if($araScrooge_eye == false){
		CreateTextureEX($araDash_Nut, $araStop_pri2, Center, Middle, "cg/ev/x/ev9002スクルージダッシュ汎用_x01.png");
	}

	CreateTextureEX($araStop_Nut, $araStop_pri, Center, 0, "cg/st/x/stスクルージ_戦闘_normal_x01.png");
	CreateTextureEX($araBack_Nut, $araStop_pri_back, Center, Middle, $araScrooge_back_file);
	CreateTextureEX($araBackback_Nut, $araStop_pri_backback, Center, Middle, $araScrooge_back_file);
	CreateMovieEX($araMovie_Nut, $araStop_pri_move, Center, Middle, true, true, "dx/mv0003集中線04_黒.ngs");

	if($araSmoke_on == true){
	CreateTextureEX($araSmoke_Nut, $araStop_pri_smoke, Center, InBottom, "cg/ef/ef_smoke04.png");
	}



//変数背景指定初期位置
	$backXposi = -(ImageHorizon($araBackback_Nut)-1024);
	$backXposi2 = -(ImageHorizon($araBackback_Nut)-1024/8);


	SetShade($araBackback_Nut, HEAVY);
	Request($ara_Nut_All, Smoothing);

	Zoom($araMovie_Nut, 0, 3000, 3000, null, true);
	Move($araBack_All, 0, $backXposi, @$backYposi, null, true);


	SetVertex($araStop_Nut, Center, Middle);
	SetVertex($araDash_Nut, Left, Middle);


	Zoom($araStop_Nut, 0, $Sczoom, $Sczoom, null, true);
	Zoom($araDash_Nut, 0, $Sczoom, $Sczoom, null, true);


//変数立ち絵指定終点位置

	if($Sczoom == 1000){
		$scrYposi = -(ImageVertical($araStop_Nut)/4)*$Sczoom/1000;
		Move($araStop_Nut, 0, @0, @$scrYposi, null, true);
	}else{
		if($Sczoom <= 1500 && $Sczoom > 1300){
			$scrYposi = -(ImageVertical($araStop_Nut)/4)*$Sczoom/1100;
			$scrXposi = -(ImageHorizon($araStop_Nut)/10)*$Sczoom/800;
			Move($araStop_Nut, 0, @$scrXposi, @$scrYposi, null, true);
		}else if($Sczoom <= 1300 && $Sczoom > 1000){
			$scrYposi = -(ImageVertical($araStop_Nut)/4)*$Sczoom/1100;
			$scrXposi = -(ImageHorizon($araStop_Nut)/20)*$Sczoom/700;
			Move($araStop_Nut, 0, @$scrXposi, @$scrYposi, null, true);
		}else if($Sczoom < 1000 && $Sczoom >= 900){
			$scrYposi = -(ImageVertical($araStop_Nut)/4)*$Sczoom/1000;
			$scrXposi = (ImageHorizon($araStop_Nut)/20)*1000/$Sczoom;
			Move($araStop_Nut, 0, @$scrXposi, @$scrYposi, null, true);
		}else if($Sczoom < 900 && $Sczoom >= 700){
			$scrYposi = -(ImageVertical($araStop_Nut)/5)*$Sczoom/1000;
			$scrXposi = (ImageHorizon($araStop_Nut)/10)*1000/$Sczoom;
			Move($araStop_Nut, 0, @$scrXposi, @$scrYposi, null, true);
		}else if($Sczoom < 700 && $Sczoom >= 501){
			$scrYposi = -(ImageVertical($araStop_Nut)/6)*$Sczoom/1200;
			$scrXposi = (ImageHorizon($araStop_Nut)/9)*1000/$Sczoom;
			Move($araStop_Nut, 0, @$scrXposi, @$scrYposi, null, true);
		}else if($Sczoom < 501 && $Sczoom >= 480){
			$scrYposi = 0;
			$scrXposi = (ImageHorizon($araStop_Nut)/9)*1000/$Sczoom;
			Move($araStop_Nut, 0, @$scrXposi, @$scrYposi, null, true);
		}else if($Sczoom < 480){
			Message("デバッグ用","そんなにちいさくてもいいのはおっぱいだけです",YESNOCANCE,EXCLAMATION);
			Delete($ara_Nut_All);
		}else{
			Message("デバッグ用","そんなにおおきいのらめえ",YESNOCANCE,EXCLAMATION);
			Delete($ara_Nut_All);
		}
	}



	Move($araStop_Nut, 0, @$ScroX, @0, null, true);


//変数ダッシュ絵指定終点位置
	Position($araStop_Nut,$ScSPosiX,$ScSPosiY);
	$ScDPosiX = $ScSPosiX + (ImageHorizon($araStop_Nut)*$Sczoom/3000);
	$ScDPosiY = $ScSPosiY + (ImageVertical($araStop_Nut)*$Sczoom/5000);

	Move($araDash_Nut, 0, $ScDPosiX, $ScDPosiY, null, true);


//変数ダッシュ絵定始点位置
	$ScDSetX = (ImageHorizon($araStop_Nut)*$Sczoom/1500));
	$ScDSetY = -(ImageVertical($araStop_Nut)*$Sczoom/8000);
	$ScDmoveX =-$ScDSetX;
	$ScDmoveY =-$ScDSetY;


	Move($araDash_Nut, 0, @$ScDSetX, @$ScDSetY, null, true);


//煙用
	$SmoPosiX = $ScSPosiX+40;
	$Smozoom = $Sczoom + 300;
	Zoom($araSmoke_Nut, 0, $Sczoom, 0, null, true);
	Move($araSmoke_Nut, 0, $SmoPosiX, @0, null, true);
	SetVertex($araSmoke_Nut, Right, bottom);

	if($Sczoom <= 1000 &&$Sczoom>700){
	$Smomove = $Sczoom/8;
	Move($araSmoke_Nut, 0, @0, @$Smomove, null, true);
	}if else($Sczoom<=700){
	}else{
		Delete($araSmoke_Nut);
	}



}

function GoArScroogeStop(){



	Fade($araMovie_Nut, 0, 1000, null, true);
	Fade($araBack_Nut, 0, 1000, null, true);
	Move($araBack_All, 700, $backXposi2, @0, Dxl3, false);
	Fade($araMovie_Nut, 400, 0, Axl1, false);
	Wait(200);
	MusicStart("@SEマクロ01",0,700,0,1000,null,false);
	Move($araDash_Nut, 250, @$ScDmoveX, @$ScDmoveY, Axl3, false);
	Wait(100);
	Fade($araDash_Nut, 50, 1000, null, false);
	Wait(100);
	Fade($araDash_Nut, 200, 0, Axl3, false);
	Wait(100);
	Move($araBack_All, 1500, @20, @0, Dxl3, false);
	Fade($araBackback_Nut, 200, 1000, Axl3, false);
	Move($araStop_Nut, 500, @-100, @-40, Dxl2, false);
	Zoom($araSmoke_Nut, 200, $Sczoom, $Smozoom, Dxl3, false);
	Fade($araSmoke_Nut, 150, 1000, null, false);
	Fade($araStop_Nut, 150, 1000, null, false);
	MusicStart("@SEマクロ02",0,700,0,1000,null,false);
	Wait(100);
	Zoom($araSmoke_Nut, 500, $Smozoom, $Sczoom, null, false);
	Fade($araSmoke_Nut, 1000, 0, null, false);
	Move($araStop_Nut, 1000, @3, @0, Dxl3, true);

}

function BlockAra_Create(){

	CreateTextureEXadd("絵破片", 1000, Center, Middle, "cg/ef/ef_結晶割れ01.png");
	CreateTextureEXadd("絵破片弐", 1000, Center, Middle, "cg/ef/ef_結晶割れ02.png");
	CreateTextureEXadd("絵破片参", 1000, Center, Middle, "cg/ef/ef_結晶割れ07.png");
	CreateTextureEXadd("絵破片四", 1000, Center, Middle, "cg/ef/ef_結晶割れ04.png");
	CreateTextureEXadd("絵破片五", 1000, Center, Middle, "cg/ef/ef_結晶割れ05.png");
	CreateTextureEXadd("絵破片六", 1000, Center, Middle, "cg/ef/ef_結晶割れ01.png");

	Request("絵破片*", Smoothing);

	SetVertex("絵破片*", right, bottom);
	SetVertex("絵破片六", right, top);

	Rotate("絵破片四", 0, @0, @0, @-20, null,true);
	Rotate("絵破片六", 0, @0, @0, @50, null,true);

}


function BlockAra_Start(){

	CreateProcess("絵破片プロ", 150, 0, 0, "BlockAra_Move");
	SetAlias("絵破片プロ","絵破片プロ");
	Request("絵破片プロ", Start);

}

function BlockAra_Move(){

	while(1){

		Move("@絵破片", 0, 225, -270, null, true);
		Move("@絵破片弐", 0, 300, 110, null, true);
		Move("@絵破片参", 0, 770, 80, null, true);
		Zoom("@絵破片", 0, 100, 100, null, true);
		Zoom("@絵破片弐", 0, 100, 100, null, true);
		Zoom("@絵破片参", 0, 100, 100, null, true);
	
	
		Move("@絵破片", 1000, -30, -525, null, false);
		Move("@絵破片弐", 1300, 30, -200, null, false);
		Move("@絵破片参", 1600, 260, -230, null, false);
		Zoom("@絵破片", 1000, 1000, 1000, null, false);
		Zoom("@絵破片弐", 1300, 1000, 1000, null, false);
		Zoom("@絵破片参", 1600, 1000, 1000, null, false);
		Fade("@絵破片", 500, 1000, null, false);
		Fade("@絵破片弐", 500, 1000, null, false);
		Fade("@絵破片参", 500, 1000, null, true);
	
		Fade("@絵破片", 500, 0, Axl3, false);
		Wait(300);
		Fade("@絵破片弐", 400, 0, Dxl1, false);
		Wait(200);
		Fade("@絵破片参", 300, 0, Dxl1, false);
	
	
		Move("@絵破片四", 0, 700, 110, null, true);
		Move("@絵破片五", 0, 400, -230, null, true);
		Move("@絵破片六", 0, 100, 700, null, true);
		Zoom("@絵破片四", 0, 0, 0, null, true);
		Zoom("@絵破片五", 0, 100, 100, null, true);
		Zoom("@絵破片六", 0, 100, 100, null, true);
	
		Move("@絵破片六", 1200, 0, -200, null, false);
		Move("@絵破片五", 1400, 0, -590, Dxl1, false);
		Move("@絵破片四", 1500, 5, -600, null, false);
		Zoom("@絵破片六", 1200, 1000, 1000, null, false);
		Zoom("@絵破片五", 1400, 600, 600, Dxl1, false);
		Zoom("@絵破片四", 1500, 900, 900, null, false);
		Fade("@絵破片四", 500, 1000, null, false);
		Fade("@絵破片五", 500, 1000, Dxl1, false);
		Fade("@絵破片六", 500, 1000, null, true);
	
		Fade("@絵破片六", 500, 0, Axl3, false);
		Wait(300);
		Fade("@絵破片五", 400, 0, Dxl1, false);
		Wait(100);
		Fade("@絵破片四", 300, 0, null, false);


	}


}

function BlockAra_Delete(){

	Request("@絵破片プロ", Stop);
	Wait(500);
	FadeDelete("@絵破片*", 500, null, true);
	Delete("@絵破片*");
}

function Fami_Text_End(){

	CreateProcess("Pro_end", 0, 0, 0, "Fami_Text_End_Act");
	SetAlias("Pro_end","Pro_end");
	Request("Pro_end", Start);

}

function Fami_Text_End_Act(){

	begin:

	while(1){

		Fade("@fami_text_end", 150, 1000, null, false);
		Wait(300);
		Fade("@fami_text_end", 150, 0, null, false);
		Wait(300);
	}


}

function Fami_Text_End_Delete(){
		Request("@fami_text_end", UnLock);
		Request("@Pro_end", UnLock);
		Delete("@fami_text_end");
		Delete("@Pro_end");
}

{
	$while=1;

		if($while==1){
			$キャラ = "蒼葉";
			$キャラpro = "1蒼葉";
		}else if($while==2){
			$キャラ = "ノイズ";
			$キャラpro = "3ノイズ";
		}else if($while==3){
			$キャラ = "蓮犬";
			$キャラpro = "6蓮犬";
		}else if($while==4){
			$キャラ = "悪島";
			$キャラpro = "8悪島";
		}

	
	//マクロキャラ用変数
		$マクロキャラ１正面 = "cg/sys/ed/dot/dm_" + $キャラpro + "_1正面01.png";
		$マクロキャラ２正面 = "cg/sys/ed/dot/dm_" + $キャラpro + "_1正面02.png";
		$マクロキャラ１後ろ = "cg/sys/ed/dot/dm_" + $キャラpro + "_2後ろ01.png";
		$マクロキャラ２後ろ = "cg/sys/ed/dot/dm_" + $キャラpro + "_2後ろ02.png";
		$マクロキャラ１左 = "cg/sys/ed/dot/dm_" + $キャラpro + "_3左01.png";
		$マクロキャラ２左 = "cg/sys/ed/dot/dm_" + $キャラpro + "_3左02.png";
		$マクロキャラ１右 = "cg/sys/ed/dot/dm_" + $キャラpro + "_4右01.png";
		$マクロキャラ２右 = "cg/sys/ed/dot/dm_" + $キャラpro + "_4右02.png";
	
	
	//優先度01・02
		$優先度１ = $優先度++;
		$優先度２ = $優先度;
	
	//プロセスナット名用変数
		$プロナット名１ = "ファミキャラ01" + $キャラ + "下";
		$プロナット名２ = "ファミキャラ02" + $キャラ + "下";
		$プロナット名３ = "ファミキャラ01" + $キャラ + "上";

		$プロナット名７ = "ファミキャラ01" + $キャラ + "右";
		$プロナット名８ = "ファミキャラ02" + $キャラ + "右";
	
	//アクションプロセス用変数
		$アクプロナット名１="@"+$プロナット名１;
		$アクプロナット名２="@"+$プロナット名２;
	
	//生成
		CreateTextureEX($プロナット名１, $優先度１, 16,16, $マクロキャラ１正面);

...//キャラムーブファンクション
function Dot_Cara_Move($キャラムーブ,$移動方向,$移動マス量,$移動待ち){

	$MposiX = 0;
	$MposiY = 0;


	$時間 = 300;
	$移動時間 = $時間*$移動マス量;



	if($移動待ち==true){
		$移動時間猶予 = $時間*$移動マス量+20;
	}else{
		$移動時間猶予 = $移動待ち;
	}

	$キャラムーブナット１ = "@ファミキャラ01"+$キャラムーブ+"*";
	$キャラムーブナット２ = "@ファミキャラ02"+$キャラムーブ+"*";

	$キャラムーブ認識用 = "@ファミキャラ01"+$キャラムーブ +"下";



	if($移動方向=="右"){
		Dot_Cara_FadeM($キャラムーブ,"右");
		Position($キャラムーブ認識用,$MPosiX,$MPosiY);
		$移動量X = $MPosiX + ($移動マス量*32);
		$移動量Y = $MPosiY;
	}else if($移動方向=="下"){
		Dot_Cara_FadeM($キャラムーブ,"下");
		Position($キャラムーブ認識用,$MPosiX,$MPosiY);
		$移動量X = $MPosiX;
		$移動量Y = $MPosiY + ($移動マス量*32);
	}else if($移動方向=="左"){
		Dot_Cara_FadeM($キャラムーブ,"左");
		Position($キャラムーブ認識用,$MPosiX,$MPosiY);
		$移動量X = $MPosiX - ($移動マス量*32);
		$移動量Y = $MPosiY;
	}else if($移動方向=="上"){
		Dot_Cara_FadeM($キャラムーブ,"上");
		Position($キャラムーブ認識用,$MPosiX,$MPosiY);
		$移動量X = $MPosiX;
		$移動量Y = $MPosiY - ($移動マス量*32);
	}


	if($移動待ち == true){
	Move($キャラムーブナット１, $移動時間, $移動量X, $移動量Y, null, false);
	Move($キャラムーブナット２, $移動時間, $移動量X, $移動量Y, null, true);
	}else{
	Move($キャラムーブナット１, $移動時間, $移動量X, $移動量Y, null, false);
	Move($キャラムーブナット２, $移動時間, $移動量X, $移動量Y, null, false);
	Wait($移動時間猶予);
	}


//	Wait(300);
}
