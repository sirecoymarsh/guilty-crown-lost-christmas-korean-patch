.//★マクロ「初期化」
function ConfigDefaultInit()
{
	//#no_ask=false;

	//★徒歩：タイトルから呼び出された場合「$OnDialog」をtrueに！
	//「sys_close関係」二重にダイアログが出るのを防止する為です
	if($PLACE_title){
		$OnDialog=true;
	}

	//★尋ねない
	if(!#no_ask){
		$go=false;

		//■定義「メッセージウインドウ作成」
		CreateName("MSGWND");
	
		//□基礎構築
		DialogBase("全ての項目を初期設定に戻しますか？");
	
		//★選択肢開始
		$SYSTEM_r_button_down=false;
		select{
			if($SYSTEM_r_button_down){break;}
	
			case MSGWND/MSG_no{
				Request("MSGWND/MSG_no/se", Play);
			}case MSGWND/MSG_yes{
				//所定の動作
				$go=true;
				Request("MSGWND/MSG_yes/se", Play);
			}
	
			//★キーダウン系
			if($SYSTEM_keydown_f){
				DialogFull();
			}else if($SYSTEM_keydown_esc||$SYSTEM_buttondown_close){
				DialogClose();
			}else if($SYSTEM_keydown_t&&!$SYSTEM_menu_lock&&!$SYSTEM_title_lock){
				DialogTitle();
			}else if($SYSTEM_keydown_r&&!$SYSTEM_menu_lock&&!$SYSTEM_backselect_lock&&ExistSave(9999)){
				DialogBack();
			}
		}
		$SYSTEM_r_button_down=false;
		$SYSTEM_menu_config_enable=true;

		if(!$go){
			//■終了共通演出
			DialogEnd();

			//★徒歩：「sys_close関係」タイトルから呼び出された場合
			$OnDialog=false;
			//MoveCursor($ConfigDefaultX_M,$ConfigDefaultY_M);
			return;
		}
	}

	//★初期化内容
	ConfigFormat();

	$SYSTEM_menu_config_enable=false;
	DialogEnd();
	$OnDialog=false;
}

					else

			case "CF_vo桜満春夏"
			{
				if(#CFGVC_HARUKA == 1)
				{

					if($CF_vo08 != 0)

				}

				}
			}

				call_chapter nss/sys_close.nss;
			}
		}
	}
